# Первый запуск (создаются тома)
docker-compose up -d

# Остановка контейнеров (тома остаются)
docker-compose down

# Повторный запуск – данные сохраняются
docker-compose up -d

# Удалить тома вместе с контейнерами (очистить данные)
docker-compose down -v

Для рестарта clickimitator внутри докер
docker compose restart clickimitator


# Пересобрать образ, так как берет его из кеша даже после очистки (если меняли код)
docker compose up --build

docker compose up -d --build storage-gateway # пересобрать конкретный сервис
